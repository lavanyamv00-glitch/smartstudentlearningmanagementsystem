package com.techpalle.springboot.assignmententity;

import com.techpalle.springboot.userentity.User;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.ManyToOne;

@Entity
public class Assignment {
	@Id
	@GeneratedValue(strategy = GenerationType.IDENTITY)
	private Long assignmentId;
	
	@ManyToOne
	@JoinColumn(name = "student_id")
	private User student;
	
	private String fileName;
	private String filePath;
	
	public Assignment() {}

	public Assignment(Long assignmentId, User student, String fileName, String filePath) {
		super();
		this.assignmentId = assignmentId;
		this.student = student;
		this.fileName = fileName;
		this.filePath = filePath;
	}

	public Long getAssignmentId() {
		return assignmentId;
	}

	public void setAssignmentId(Long assignmentId) {
		this.assignmentId = assignmentId;
	}
	
	public User getStudent() {
		return student;
	}

	public void setStudent(User student) {
		this.student = student;
	}

	public String getFileName() {
		return fileName;
	}

	public void setFileName(String fileName) {
		this.fileName = fileName;
	}

	public String getFilePath() {
		return filePath;
	}

	public void setFilePath(String filePath) {
		this.filePath = filePath;
	}
	
	
}

package com.techpalle.springboot.assignmentservice;

import java.io.File;
import java.io.IOException;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;
import com.techpalle.springboot.assignmentdto.AssignmentResponseDTO;
import com.techpalle.springboot.assignmententity.Assignment;
import com.techpalle.springboot.assignmentrepository.AssignmentRepository;
import com.techpalle.springboot.userentity.User;
import com.techpalle.springboot.userrepository.UserRepository;

@Service
public class AssignmentService {
	
	private final static String UPLOAD_DIR = "uploads/";
	private final AssignmentRepository assignmentRepository;
	private final UserRepository userRepository;
	
	@Autowired
	public AssignmentService(AssignmentRepository assignmentRepository,UserRepository userRepository) {
		super();
		this.assignmentRepository = assignmentRepository;
		this.userRepository = userRepository;
	}
	
	public AssignmentResponseDTO uploadFile(MultipartFile file,Long studentId) throws IOException {
		User student = userRepository.findById(studentId).
				orElseThrow(() -> new RuntimeException("student not found"));
		
	    String fileName = file.getOriginalFilename();
	    String filePath = UPLOAD_DIR + fileName;

	    File dest = new File(filePath);
	    dest.getParentFile().mkdir();
	    file.transferTo(dest);

	    Assignment assignment = new Assignment();
	    assignment.setFileName(fileName);
	    assignment.setFilePath(filePath);
	    assignment.setStudent(student);

	    Assignment saved = assignmentRepository.save(assignment);
	    
	    //convert Entity --> DTO
	    return new AssignmentResponseDTO(
	    		saved.getAssignmentId(),
	    		saved.getStudent().getSname(),
	    		saved.getFileName(),
	    		saved.getFilePath()
	    		);
	}
	
	public String deleteFile(Long id) {

	    Assignment assignment = assignmentRepository.findById(id)
	            .orElseThrow(() -> new RuntimeException("File not found"));
	    File file = new File(assignment.getFilePath());
	    if (file.exists()) {
	        file.delete();
	    }
	    assignmentRepository.delete(assignment);
	    return "File deleted successfully";
	}
}

package com.techpalle.springboot.assignmentcontroller;

import java.io.IOException;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.multipart.MultipartFile;
import com.techpalle.springboot.assignmententity.Assignment;
import com.techpalle.springboot.assignmentservice.AssignmentService;
import com.techpalle.springboot.userentity.User;

@RestController
@RequestMapping("/api/assignment")
public class AssignmentController {
	
	private final AssignmentService assignmentService;
	
	@Autowired
	public AssignmentController(AssignmentService assignmentService) {
		super();
		this.assignmentService = assignmentService;
	}
	
	@PostMapping("/uploadfile")
	public AssignmentResponseDTO uploadFile(@RequestParam("file") MultipartFile file,
			     @RequestParam("studentId") User studentId) throws IOException {
		return assignmentService.uploadFile(file,studentId);
	}
	@DeleteMapping("/deletefile")
	public String deleteFile(@PathVariable Long id) {
		assignmentService.deleteFile(id);
		return "File deleted successfully";
	}
}
